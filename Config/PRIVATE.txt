# 私有配置覆盖：由 Settings.sh 追加到 .config 末尾。
# kconfig 解析 .config 时同名 CONFIG 以最后一次赋值为准。
#
# 1) HomeProxy 与 Nikki 共存
#    GENERAL.txt 里已有 luci-app-homeproxy=y，这里不再覆盖；
#    homeproxy 源码来自 VIKINGYFY/packages（Packages.sh 的 UPDATE_PACKAGE "viking" 克隆），
#    依赖的 sing-box 同样来自该仓库，版本配套。
#    nikki 用 mihomo 内核，homeproxy 用 sing-box 内核，两者互不冲突。
#
# 2) Nikki（nikkinikki-org/OpenWrt-nikki，由 Packages.sh 的 UPDATE_PACKAGE 克隆）
#    nikki                  核心（DEPENDS 带 kmod-tun / kmod-nft-tproxy / ip-full 等）
#    luci-app-nikki         LuCI 界面（DEPENDS +nikki）
#    luci-i18n-nikki-zh-cn  中文语言包（由 luci.mk 从 po/zh_Hans 生成）
CONFIG_PACKAGE_nikki=y
CONFIG_PACKAGE_luci-app-nikki=y
CONFIG_PACKAGE_luci-i18n-nikki-zh-cn=y
#
# 3) mihomo 内核二选一：mihomo-meta（稳定版）/ mihomo-alpha（测试版），
#    两者都 PROVIDES:=mihomo，nikki 依赖的就是这个虚拟名 mihomo。
#    mihomo-meta 自带 DEFAULT_VARIANT:=1，这里再显式钉住，避免变体默认值以后变化。
CONFIG_PACKAGE_mihomo-meta=y
